import numpy as np
from sklearn.datasets import load_iris
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import GridSearchCV, train_test_split

# 1. Load data and split into train/test sets
data = load_iris()
X_train, X_test, y_train, y_test = train_test_split(
    data.data, data.target, test_size=0.2, random_state=42
)

# 2. Initialize the base estimator (model)
rf = RandomForestClassifier(random_state=42)

# 3. Define the parameter grid (values to try)
param_grid = {
    'n_estimators': [50, 100, 200],
    'max_depth': [None, 10, 20],
    'min_samples_split': [2, 5]
}

# 4. Set up the GridSearchCV object (3 x 3 x 2 = 18 combinations; 5 folds each)
# n_jobs=-1 utilizes all available processor cores to speed up the process
grid_search = GridSearchCV(estimator=rf, param_grid=param_grid, cv=5, scoring='accuracy', n_jobs=-1)

# 5. Fit the model to the training data
grid_search.fit(X_train, y_train)

# 6. Extract the best performance metrics
print("Best Hyperparameters:", grid_search.best_params_)
print("Best Cross-Validation Score:", grid_search.best_score_)

# 7. Use the optimized model to predict on new data
best_model = grid_search.best_estimator_
test_accuracy = best_model.score(X_test, y_test)
print("Test Set Accuracy:", test_accuracy)
