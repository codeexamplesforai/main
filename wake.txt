#!/usr/bin/env python3
"""
keep_awake_noperm.py — Keeps your Mac awake (system, display, disk, network)
using only the built-in `caffeinate` tool. No Accessibility access, no admin,
no third-party packages required.

Usage:
    python3 keep_awake_noperm.py              # stay awake until Ctrl+C
    python3 keep_awake_noperm.py 7200         # stay awake for 2 hours (seconds)

To tie wakefulness to a specific command instead, skip this script and run:
    caffeinate -dims python3 my_agent_flow.py
"""

import subprocess
import sys
import signal
import time

# -d display sleep, -i idle system sleep, -m disk sleep, -s system sleep on AC
FLAGS = "-dims"

timeout = sys.argv[1] if len(sys.argv) > 1 else None
cmd = ["caffeinate", FLAGS] + (["-t", timeout] if timeout else [])

proc = subprocess.Popen(cmd)


def stop(*_):
    proc.terminate()
    print("\nReleased — your Mac can sleep normally again.")
    sys.exit(0)


signal.signal(signal.SIGINT, stop)
signal.signal(signal.SIGTERM, stop)

print(f"Mac will stay awake{' for ' + timeout + 's' if timeout else ' until you press Ctrl+C'}.")
print("Network connections will stay alive as long as the system doesn't sleep.")

while proc.poll() is None:
    time.sleep(5)

print("caffeinate exited.")
