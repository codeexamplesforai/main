#!/usr/bin/env bash
set -uo pipefail

PARENT="${1:-$PWD}"
LOG="$PARENT/graphify-batch.log"
: > "$LOG"

for dir in "$PARENT"/*/; do
  repo="${dir%/}"
  name="$(basename "$repo")"

  # only real repos
  [ -d "$repo/.git" ] || { echo "skip (no .git): $name" | tee -a "$LOG"; continue; }

  # resumable: don't redo finished ones
  if [ -f "$repo/graphify-out/graph.json" ]; then
    echo "skip (done): $name" | tee -a "$LOG"; continue
  fi

  echo "=== $name ===" | tee -a "$LOG"
  if ( cd "$repo" && graphify extract . --global --as "$name" --max-workers 4 ) >>"$LOG" 2>&1; then
    echo "ok: $name" | tee -a "$LOG"
  else
    echo "FAILED: $name" | tee -a "$LOG"
  fi
done

graphify global list | tee -a "$LOG"
